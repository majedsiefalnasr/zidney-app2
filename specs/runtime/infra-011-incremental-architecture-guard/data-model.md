## 1. Dependency Graph JSON Schema

### 1.1 Schema v1 (Existing — to be migrated)

File: `docs/ai/context/ai-dependency-graph.json`

```typescript
// Current (v1) — read-only compatible
interface DependencyGraphV1 {
  nodes: string[]; // e.g. ["packages/logger", "apps/api"]
  edges: Array<{ from: string; to: string }>; // may contain duplicates
}
```

**Limitations:**

- No `schema_version` (cannot detect migration need)
- No `generated_at` (cannot check cache freshness)
- Contains duplicate edges (same `from|to` pair repeated per import statement)
- No `reverse_dependencies` (BFS traversal requires full edge-scan on every call)

### 1.2 Schema v2 (Target — required by incremental guard)

```typescript
/**
 * Dependency graph schema v2.
 * Generated by: scripts/infra-audit.ts --generate-graph
 * Consumed by:  scripts/ai-guard.ts --incremental
 *
 * IMPORTANT: Do not edit this file manually.
 * To regenerate: bun scripts/infra-audit.ts --generate-graph
 */
interface DependencyGraph {
  /** Schema version. Must equal 2 for the incremental guard to accept this file. */
  schema_version: 2;

  /** ISO-8601 timestamp of when this file was generated. Used for cache freshness checks. */
  generated_at: string;

  /**
   * Ordered list of all declared module paths.
   * Source: Object.keys(ARCHITECTURE_MAP.json#modules)
   */
  modules: string[];

  /**
   * Deduplicated list of directed dependencies.
   * `from` imports from `to`.
   * One entry per unique module-pair (not per import statement).
   */
  edges: DependencyEdge[];

  /**
   * Pre-computed reverse adjacency list.
   * Key: module path.
   * Value: list of modules that import from this module.
   * Used by BFS traversal in compute_impact_scope().
   */
  reverse_dependencies: Record<string, string[]>;
}

interface DependencyEdge {
  /** The importing module (e.g. "apps/api") */
  from: string;
  /** The imported module (e.g. "packages/logger") */
  to: string;
}
```

**Example v2 document:**

```json
{
  "schema_version": 2,
  "generated_at": "2026-03-10T14:30:00.000Z",
  "modules": [
    "packages/job-queue",
    "packages/logger",
    "packages/types",
    "packages/config",
    "packages/redis-utils",
    "packages/ui-system",
    "packages/api-client",
    "packages/domain-core",
    "packages/validation",
    "apps/mmc",
    "apps/frontoffice",
    "apps/backoffice",
    "apps/api",
    "apps/worker"
  ],
  "edges": [
    { "from": "packages/job-queue", "to": "packages/logger" },
    { "from": "packages/redis-utils", "to": "packages/logger" },
    { "from": "packages/domain-core", "to": "packages/logger" },
    { "from": "packages/domain-core", "to": "packages/types" },
    { "from": "packages/domain-core", "to": "packages/validation" },
    { "from": "apps/api", "to": "packages/domain-core" },
    { "from": "apps/api", "to": "packages/logger" },
    { "from": "apps/api", "to": "packages/config" },
    { "from": "apps/worker", "to": "packages/domain-core" },
    { "from": "apps/worker", "to": "packages/job-queue" }
  ],
  "reverse_dependencies": {
    "packages/logger": [
      "packages/job-queue",
      "packages/redis-utils",
      "packages/domain-core",
      "apps/api"
    ],
    "packages/types": ["packages/domain-core"],
    "packages/validation": ["packages/domain-core"],
    "packages/domain-core": ["apps/api", "apps/worker"],
    "packages/config": ["apps/api"],
    "packages/job-queue": ["apps/worker"]
  }
}
```

**Schema constraints:**

- `modules` MUST match `Object.keys(ARCHITECTURE_MAP.json#modules)` at generation time
- `edges` MUST be deduplicated: at most one entry per unique `{from, to}` pair
- `reverse_dependencies` MUST be derived from `edges` — no additional entries
- All `from` and `to` values in `edges` MUST be present in `modules`
- Module paths use `/` separator, never `\`
- Module paths are always 2-segment: `apps/<name>` or `packages/<name>`

## 2. Architecture Impact Report JSON Schema

### 2.1 TypeScript Interface

```typescript
/**
 * Architecture validation result produced by ai-guard.ts on every run.
 * Written to stdout as structured output.
 * Written to docs/ai/context/architecture-impact-report.json in CI only.
 */
interface ArchitectureImpactReport {
  /** ISO-8601 generation timestamp */
  generated_at: string;

  /** Validation mode */
  mode: "incremental" | "full";

  /** Git context at time of validation */
  git_context: GitContext;

  /** Impact analysis results */
  analysis: ImpactAnalysis;

  /** Validation outcome */
  validation: ValidationOutcome;
}

interface GitContext {
  /** Current branch name */
  branch: string;

  /** Short commit hash. Empty string if not yet committed (pre-commit). */
  commit_hash: string;

  /** Number of staged/changed files that triggered validation */
  changed_files_count: number;
}

interface ImpactAnalysis {
  /** Raw list of staged file paths */
  staged_files: string[];

  /** Modules directly mapped from staged files */
  changed_modules: string[];

  /** Additional modules added via reverse-dependency traversal */
  affected_modules: string[];

  /** Union: changed_modules ∪ affected_modules — the set that was validated */
  expanded_scope: string[];

  /**
   * How many declared modules were NOT validated.
   * Equals: total_modules - expanded_scope.length
   * Always 0 in full mode.
   */
  modules_skipped: number;

  /**
   * File paths that did not match any module prefix.
   * E.g. docs/, scripts/, .husky/ — silently skipped.
   */
  skipped_unmapped_files: string[];

  /**
   * If full fallback was triggered, the reason.
   * Null if incremental validation ran successfully.
   */
  fallback_reason: FallbackReason | null;
}

type FallbackReason =
  | "map_changed" // ARCHITECTURE_MAP.json was staged
  | "graph_missing" // ai-dependency-graph.json not found
  | "graph_stale" // Graph age exceeded ARCH_GRAPH_MAX_AGE_HOURS
  | "graph_corrupt" // Graph schema version mismatch or parse error
  | "new_module" // New apps/* or packages/* dir detected, not in map
  | "full_scope"; // Computed scope = all modules (no efficiency gain)

interface ValidationOutcome {
  /** Overall pass/fail verdict */
  verdict: "PASS" | "BLOCKED";

  /** Detailed violation list. Empty if verdict = PASS. */
  violations: ArchitectureViolation[];

  /** Total number of architecture rules checked across all validated modules */
  rules_checked: number;

  /** Wall-clock duration of the validation run in milliseconds */
  duration_ms: number;
}

interface ArchitectureViolation {
  /** The module that contains the violation */
  module: string;

  /** The source file where the violation was found */
  file: string;

  /** Human-readable description of the violation */
  message: string;

  /**
   * Rule category that was violated.
   * Matches existing ai-guard.ts rule categories.
   */
  rule: ViolationRule;

  /** Severity (all violations are currently ERROR — WARN reserved for future) */
  severity: "ERROR" | "WARN";
}

type ViolationRule =
  | "FORBIDDEN_DEPENDENCY" // Importing a forbidden module
  | "LAYER_VIOLATION" // Importing from a higher/disallowed layer
  | "CROSS_APP_IMPORT" // apps/* importing another apps/*
  | "RELATIVE_LEAK" // Relative path escaping module boundary
  | "CONTRACT_VIOLATION"; // Import not declared in ARCHITECTURE_CONTRACT.json
```

## 3. Internal TypeScript Types

### 3.1 Guard Configuration (parsed CLI args)

```typescript
/**
 * Parsed configuration from CLI arguments and environment variables.
 * Created by parseArgs() at the start of main().
 */
interface GuardConfig {
  /**
   * Validation mode.
   * - 'full': scan all modules in ARCHITECTURE_MAP.json
   * - 'incremental': scan only modules affected by STAGED_FILES
   */
  mode: "full" | "incremental";

  /**
   * Explicit module list override.
   * Set by --modules flag. Bypasses staged-file detection.
   * Null = use auto-detection from STAGED_FILES.
   */
  explicitModules: string[] | null;

  /**
   * Output format.
   * - false: human-readable console output
   * - true: JSON impact report to stdout
   */
  outputJson: boolean;
}
```

### 3.2 Module Mapping Result

```typescript
/**
 * Result of mapping staged file paths to architecture module identifiers.
 * Produced by mapToModules().
 */
interface ModuleMappingResult {
  /** Modules directly touched by the staged files */
  modules: Set<string>;

  /**
   * Files that did not match any module prefix.
   * Silently skipped — not an error condition.
   * Included in the impact report for observability.
   */
  skipped: string[];
}
```

### 3.3 Scope Resolution Result

```typescript
/**
 * Complete impact scope after BFS traversal of reverse dependencies.
 * Produced by computeImpactScope().
 */
interface ScopeResolutionResult {
  /** Modules directly mapped from staged files */
  directlyChanged: Set<string>;

  /** Modules added via reverse-dependency expansion */
  expanded: Set<string>;

  /** Union: directlyChanged ∪ expanded — what gets validated */
  fullScope: Set<string>;

  /**
   * Whether the scope equals all declared modules.
   * When true, incremental is equivalent to full — use full scan.
   */
  isFullScope: boolean;
}
```

### 3.4 Validation Config (per-module)

```typescript
/**
 * Per-module validation context passed to the existing validation routines.
 * Adapts incremental calls to use the same internal validation API.
 */
interface ModuleValidationConfig {
  /** The module being validated, e.g. "apps/api" */
  modulePath: string;

  /** Architecture map for this module's rules */
  archMap: ArchitectureMap;

  /** Module boundaries for layer checks */
  boundaries: ModuleBoundaries | null;

  /** TypeScript path aliases for import resolution */
  aliases: TsAliasMap[];
}
```

### 3.5 Fallback Context

```typescript
/**
 * Context object passed to runFull() when triggered as a smart fallback
 * from runIncremental().
 */
interface FallbackContext {
  /**
   * The reason the incremental guard escalated to full scan.
   * Used in the impact report.
   */
  fallbackReason?: FallbackReason;

  /**
   * The staged files that were being analyzed when fallback was triggered.
   * Included in the impact report for traceability.
   */
  stagedFiles?: string[];
}
```
